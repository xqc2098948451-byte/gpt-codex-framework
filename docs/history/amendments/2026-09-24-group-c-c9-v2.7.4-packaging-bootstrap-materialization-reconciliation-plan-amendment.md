def _validate_schema(instance: Any, schema: dict[str, Any], root: dict[str, Any]) -> list[str]:
    errors: list[str] = []
    if "$ref" in schema:
        return _validate_schema(instance, root["$defs"][schema["$ref"].rsplit("/", 1)[-1]], root)
    allowed_types = schema.get("type")
    if allowed_types is not None:
        allowed = allowed_types if isinstance(allowed_types, list) else [allowed_types]
        matches = {"object": isinstance(instance, dict), "array": isinstance(instance, list), "string": isinstance(instance, str), "integer": isinstance(instance, int) and not isinstance(instance, bool), "null": instance is None}
        if not any(matches.get(kind, False) for kind in allowed):
            return [f"type mismatch: expected {allowed}"]
    if "const" in schema and instance != schema["const"]:
        errors.append(f"const mismatch: expected {schema['const']!r}")
    if isinstance(instance, str):
        if len(instance) < schema.get("minLength", 0): errors.append("string shorter than minLength")
        if "pattern" in schema and not __import__("re").search(schema["pattern"], instance): errors.append("string fails pattern")
    if isinstance(instance, int) and not isinstance(instance, bool) and instance < schema.get("minimum", instance): errors.append("integer smaller than minimum")
    if isinstance(instance, dict):
        for key in schema.get("required", []):
            if key not in instance: errors.append(f"missing required property: {key}")
        properties = schema.get("properties", {})
        if schema.get("additionalProperties") is False:
            errors.extend(f"unexpected property: {key}" for key in instance if key not in properties)
        for key, value in instance.items():
            if key in properties:
                errors.extend(f"{key}: {error}" for error in _validate_schema(value, properties[key], root))
    if isinstance(instance, list):
        if len(instance) < schema.get("minItems", 0): errors.append("array shorter than minItems")
        if len(instance) > schema.get("maxItems", len(instance)): errors.append("array longer than maxItems")
        if schema.get("uniqueItems") and len({json.dumps(item, sort_keys=True) for item in instance}) != len(instance): errors.append("array items are not unique")
        if "items" in schema:
            for index, item in enumerate(instance): errors.extend(f"{index}: {error}" for error in _validate_schema(item, schema["items"], root))
    return errors
