### Common Work Unit builder

```python
import re

RELEASE_SOURCE_PATHS = (
    "VERSION",
    ".gpt-codex/builtins/INDEX.json",
    ".gpt-codex/CHANGELOG.md",
    "AGENTS.md",
    ".gpt-codex/BOOTSTRAP_PROMPT.md",
    ".gpt-codex/README.md",
)

PACKAGING_PATHS = (
    ".gpt-codex/release/consumer-projection-manifest.json",
    "dist/gpt-codex-framework-v2.7.3-bootstrap.zip",
    "dist/gpt-codex-framework-v2.7.3-bootstrap.zip.sha256",
    "dist/gpt-codex-framework-v2.7.3-release.json",
    "dist/gpt-codex-framework-v2.7.4-bootstrap.zip",
    "dist/gpt-codex-framework-v2.7.4-bootstrap.zip.sha256",
    "dist/gpt-codex-framework-v2.7.4-release.json",
    "releases/INDEX.json",
    "releases/records/v2.7.4.json",
)

def require_sha40(value, code):
    if not isinstance(value, str) or not re.fullmatch(r"[0-9a-f]{40}", value):
        raise ValueError(code)
    return value

def require_sha256(value, code):
    if not isinstance(value, str) or not re.fullmatch(r"[0-9a-f]{64}", value):
        raise ValueError(code)
    return value

def canonical_source_identities(raw):
    if set(raw) != set(RELEASE_SOURCE_PATHS):
        raise ValueError("SOURCE_IDENTITIES_PATH_SET_MISMATCH")
    result = []
    for path in RELEASE_SOURCE_PATHS:
        identity = raw[path]
        if set(identity) != {"sha256", "git_blob_sha1"}:
            raise ValueError("SOURCE_IDENTITY_SHAPE_MISMATCH")
        result.append({
            "path": path,
            "sha256": require_sha256(identity["sha256"], "INVALID_SOURCE_SHA256"),
            "git_blob_sha1": require_sha40(identity["git_blob_sha1"], "INVALID_SOURCE_GIT_BLOB_SHA1"),
        })
    return result

def canonical_packaging_identities(raw):
    if set(raw) != set(PACKAGING_PATHS):
        raise ValueError("PACKAGING_IDENTITIES_PATH_SET_MISMATCH")
    result = []
    for path in PACKAGING_PATHS:
        identity = raw[path]
        if set(identity) != {"operation", "sha256", "git_blob_sha1"}:
            raise ValueError("PACKAGING_IDENTITY_SHAPE_MISMATCH")
        operation = identity["operation"]
        sha256 = identity["sha256"]
        blob = identity["git_blob_sha1"]
        if operation not in {"CREATE", "MODIFY", "DELETE"}:
            raise ValueError("INVALID_PACKAGING_OPERATION")
        if operation == "DELETE":
            if sha256 is not None or blob is not None:
                raise ValueError("DELETE_IDENTITY_MUST_HAVE_NULL_POSTIMAGE")
        else:
            sha256 = require_sha256(sha256, "INVALID_PACKAGING_SHA256")
            blob = require_sha40(blob, "INVALID_PACKAGING_GIT_BLOB_SHA1")
        result.append({"path": path, "operation": operation, "sha256": sha256, "git_blob_sha1": blob})
    return result

            "fitness": [],
        },
        "permissions": {
            "authorized_actions": ["READ", "TEST", "VALIDATE", "REPORT", "MUTATE_APPROVED_SCOPE"],
            "forbidden_actions": ["COMMIT", "PUSH", "PUBLISH", "AUTHORIZE", "SCOPE_EXPANSION"],

        "state": "AUTHORIZED",

        "artifact_refs": {
```
