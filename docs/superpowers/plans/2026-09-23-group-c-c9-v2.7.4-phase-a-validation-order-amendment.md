### 12.3 Exact six-path source content-set algorithm

```python
RELEASE_SOURCE_PATHS = (
    "VERSION", ".gpt-codex/builtins/INDEX.json", ".gpt-codex/CHANGELOG.md",
    "AGENTS.md", ".gpt-codex/BOOTSTRAP_PROMPT.md", ".gpt-codex/README.md",
)

def canonical_source_identities(raw):
    if set(raw) != set(RELEASE_SOURCE_PATHS):
        raise ValueError("SOURCE_IDENTITY_PATH_SET_MISMATCH")
    return [
        {"path": path, "sha256": raw[path]["sha256"],
         "git_blob_sha1": raw[path]["git_blob_sha1"]}
        for path in RELEASE_SOURCE_PATHS
    ]

import hashlib
import json

def canonical_source_content_set_bytes(raw):
    identities = canonical_source_identities(raw)
    return (
        json.dumps(identities, ensure_ascii=False, sort_keys=True, indent=2)
        + "\\n"
    ).encode("utf-8")

def source_content_set_sha256(raw):
    return hashlib.sha256(canonical_source_content_set_bytes(raw)).hexdigest()
```
